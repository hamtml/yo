function descargarDOCX(){
    let d=recolectarDatos();
    const{Document,Packer,Paragraph,TextRun,Table,TableRow,TableCell,WidthType,HeadingLevel}=docx;
    function celda(t,b=false,w=4000){return new TableCell({width:{size:w,type:WidthType.DXA},children:[new Paragraph({children:[new TextRun({text:String(t),bold:b,size:20,font:'Segoe UI'})]})]});}
    function fila(l,v){return new TableRow({children:[celda(l,true,3500),celda(v,false,6500)]});}
    let ap=(d.apelantes||[]).join(', ')||'Ninguno',ad=d.hayAdhesivos==='si'?((d.adhesivos||[]).join(', ')||'Ninguno'):'No hay';
    let delText=(d.delitosDatos||[]).join(', ')||'—';
    let children=[new Paragraph({children:[new TextRun({text:'REPORTE DE TOCA PENAL',bold:true,size:36,font:'Segoe UI',color:'0f172a'})]}),new Paragraph({children:[new TextRun({text:'Toca: '+(d.toca||'—'),bold:true,size:28,color:'2563eb'})]}),new Paragraph({spacing:{after:300}}),new Table({width:{size:10000,type:WidthType.DXA},rows:[fila('Sala',d.sala||'—'),

fila('Acto impugnado',(d.actoImpugnado||'—')+' ('+fechaLegible(d.fechaActo)+')')
,fila('Delito(s)',delText),fila('Apelantes principales',ap),fila('Apelantes adhesivos',ad),fila('Procedencia',d.distrito||'—'),fila('Circuito',d.circuito||'—'),fila('Turno',fechaLegible(d.turno)),

fila('Volumen',(d.volumenDiscos||'—')+' discos')
]})];
    let doc=new Document({sections:[{children}]});
    Packer.toBlob(doc).then(blob=>{saveAs(blob,'reporte_toca_'+(d.toca||'sinToca')+'.docx');});
}
